class Account
    private balance as float
    protected accountNumber as string
    public owner as string
    
    public sub new()
        m.balance = 0.0
    end sub
    
    private function validateAmount(amount as float) as boolean
        return amount > 0
    end function
    
    public function deposit(amount as float) as void
        if m.validateAmount(amount) then
            m.balance = m.balance + amount
        end if
    end function
end class

class Shape
    function getArea() as float
        return 0.0
    end function
end class

class Circle extends Shape
    radius as float
    
    override function getArea() as float
        return 3.14159 * m.radius * m.radius
    end function
end class
